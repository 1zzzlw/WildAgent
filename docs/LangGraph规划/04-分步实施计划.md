class GenerationState(TypedDict):

    user_message: str
    building_type: NotRequired[str]  # 可选，从user_message推断
    
    # ── Layer 0: 骨架 ──

    skeleton_summary: NotRequired[str]     # 供后续节点注入
    wall_bounding_box: NotRequired[dict]   # get_wall_bounding_box结果
    
    # ── Layer 1: 分片（并行）──
    door_fragments: NotRequired[list[dict]]
    window_fragments: NotRequired[list[dict]]
    # roof不是组件，直接在skeleton中生成或Layer1单独处理
    canopy_fragments: NotRequired[list[dict]]
    balcony_fragments: NotRequired[list[dict]]
    railing_fragments: NotRequired[list[dict]]
    ramp_fragments: NotRequired[list[dict]]
    bay_window_fragments: NotRequired[list[dict]]
    cornice_fragments: NotRequired[list[dict]]
    chimney_fragments: NotRequired[list[dict]]
    light_fragments: NotRequired[list[dict]]
    
    # ── Layer 2: 合并 + 校验 ──
    merged_blueprint: NotRequired[dict]
    validation_results: NotRequired[list[dict]]  # PipelineStepResult列表
    failed_components: NotRequired[list[dict]]   # 结构化失败信息
    passed_component_ids: NotRequired[list[str]]
    retry_count: int
    max_retries: int
    
    # ── 回调 ──
    callback_context: NotRequired[dict]
    
    # ── 最终输出 ──
    final_blueprint: NotRequired[dict]
    error: NotRequired[str]
    status: str  # "complete" | "partial" | "failed"
```

def build_generation_graph() -> StateGraph:
    graph = StateGraph(GenerationState)

    graph.add_node("skeleton", skeleton_generator)

    # Layer 1 (并行)
    graph.add_node("doors", door_generator)

    graph.add_node("roof", roof_generator)

> 预期：skeleton 完成后，doors/windows/roof 三个节点并行执行

### Step 2.3: 逐组件添加

按优先级逐个实现组件节点（基于 `wild-compiler/componentRegistry.ts` 实际注册的 10 种）：

| 优先级 | 组件 | 编译器位置 | 原因 |
|---|---|---|---|
| P0 | door, window, roof | `components/door.ts`, `window.ts` | 建筑三要素 |
| P1 | railing | `components/railing.ts` | 常见附属，被 balcony 复用 |
| P2 | canopy, balcony | `components/canopy.ts`, `balcony.ts` | 装饰型，balcony 内嵌 railing |
| P3 | light | `components/light.ts` | 交互型，生成 behaviors.interactions |
| P4 | ramp, bay_window | `components/ramp.ts`, `bayWindow.ts` | 特殊场景 |
| P5 | cornice, chimney | `components/cornice.ts`, `chimney.ts` | 中式建筑特有 |

**实现顺序严格遵守依赖关系：** railing 必须在 balcony 之前，因为 balcony 编译器内部调用 railing 编译逻辑。

**每个组件节点验证后才加下一个，不批量实现。**

### Step 3.1: 合并节点

async def merge_fragments_node(state: GenerationState) -> dict:
    """收集所有 Layer 1 产出 → merge_fragments() → merged_blueprint"""
    fragments = []
    for key in ["door_fragments", "window_fragments", "roof_fragment", ...]:
        if state.get(key):
            fragments.append(...)
    merged = merge_fragments(state["skeleton_blueprint"], fragments)
    return {"merged_blueprint": merged}
```

### Step 3.2: 校验节点

```python
async def validate_node(state: GenerationState) -> dict:
    """直接复用 run_validation_pipeline()"""
    results = run_validation_pipeline(state["merged_blueprint"])
    errors = _final_errors(results)
    # 将 error 追溯到具体 component
    failed = trace_errors_to_components(errors, state["merged_blueprint"])
    passed = list(set(get_all_component_ids(state)) - set(f["component_id"] for f in failed))
    return {
        "validation_results": results,
        "failed_components": failed,
        "passed_component_ids": passed
    }
```

### Step 3.3: 条件路由——重试 or 完成

```python
def should_retry(state: GenerationState) -> str:
    if not state["failed_components"]:
        return "done"
    if state["retry_count"] >= state["max_retries"]:
        return "escalate"
    return "retry"

graph.add_conditional_edges(
    "validate",
    should_retry,
    {
        "done": "finish",
        "retry": "callback",
        "escalate": "escalate_handler"
    }
)
```

### Step 3.4: 回调节点

```python
async def callback_node(state: GenerationState) -> dict:
    """对每个失败的组件：RAG + tools → callback_payload → LLM 修正"""
    payload = build_callback_payload(state)
    # 只修正失败的组件
    fixed_fragments = await fix_failed_components(payload, state)
    # 更新对应的 fragments
    return update_fragments(state, fixed_fragments)
```

**验证**：

> 故意构造一个含错误的 Blueprint（door from[0] 超出墙体）
> 预期：validate → 检测到错误 → callback → LLM 修正 → recheck 通过

## 六、风险与缓解

| 风险 | 概率 | 缓解 |
|---|---|---|
| LLM 调用次数 ×5~10，token 费用翻倍 | 高 | 小模型负责简单组件（furniture/railing），大模型负责复杂组件（roof） |
| 并行节点互相竞争 LLM API 限流 | 中 | 信号量控制并发 LLM 调用数 |
| 回调循环无限重试 | 低 | max_retries=3 硬限制 + 降级处理 |
| 分片之间引用断裂 | 中 | merge 阶段的引用完整性检测 + ID 前缀约定 |
| 骨架生成失败 → 后续节点空转 | 中 | 骨架失败直接短路（不进入 Layer 1） |
