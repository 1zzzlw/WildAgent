# LangGraph 可观测性与调试设计

> 前端展示完整的 RAG 日志、思考内容、性能指标

---

## 一、需求总结

### 1.1 用户需求

前端需要展示：
1. **RAG 检索日志**：每个节点的 RAG 查询、命中的 chunk、metadata 过滤条件
2. **节点思考内容**：每个节点的推理过程（如果使用思考模型）
3. **性能指标**：
   - Token 消耗量（输入/输出分开）
   - 耗时（总耗时 + 每个节点耗时）
   - RAG 检索次数和命中率
   - 重试次数
4. **调试信息**：节点状态、错误堆栈、校验结果

### 1.2 使用场景

- **开发调试**：定位 RAG 召回不准确、某个节点生成错误
- **用户反馈**：用户报告问题时，导出完整日志
- **性能优化**：分析哪个节点最慢、Token 消耗在哪里
- **知识库质量**：检查哪些 chunk 被召回、是否相关

### 2.2 后端发送实现


```python
# wild-server/app/agent/debug_logger.py
import time
from typing import Any

class DebugLogger:
    """LangGraph 节点执行的可观测性日志收集器"""
    
    def __init__(self, request_id: str, ws_send_fn):
        self.request_id = request_id
        self.ws_send = ws_send_fn
        self.start_time = time.time()
        self.node_timings: dict[str, float] = {}
        self.token_usage: dict[str, dict] = {}
        self.rag_stats = {
            "queries": 0,
            "cache_hits": 0,
            "total_chunks": 0,
        }
    
    async def log_rag(
        self,
        node: str,
        query: str,
        filter: dict,
        hits: list,
        duration_ms: float,
        cache_hit: bool = False,
    ):
        """记录 RAG 检索日志"""
        self.rag_stats["queries"] += 1
        if cache_hit:
            self.rag_stats["cache_hits"] += 1
        self.rag_stats["total_chunks"] += len(hits)
        
        # 发送给前端
        await self.ws_send({
            "type": "debug_log",
            "request_id": self.request_id,
            "category": "rag",
            "timestamp": int(time.time() * 1000),
            "data": {
                "node": node,
                "query": query[:200],  # 截断长查询
                "filter": filter,
                "hits": [
                    {
                        "chunk_id": hit.metadata.get("chunk_index", "?"),
                        "score": hit.distance if hasattr(hit, "distance") else 0,
                        "source": hit.metadata.get("source", "?"),
                        "heading": hit.metadata.get("heading", "?"),
                        "doc_type": hit.metadata.get("doc_type", "?"),
                        "entity_type": hit.metadata.get("entity_type", "?"),
                        "status": hit.metadata.get("status", "?"),
                        "preview": hit.document[:100],
                    }
                    for hit in hits[:5]  # 只发前 5 个
                ],
                "duration_ms": duration_ms,
                "cache_hit": cache_hit,
            }
        })
    
    async def log_node_stage(
        self,
        node: str,
        stage: str,
        message: str,
        token_usage: dict | None = None,
    ):
        """记录节点阶段"""
        if stage == "start":
            self.node_timings[node] = time.time()
        elif stage in ("complete", "failed"):
            duration = time.time() - self.node_timings.get(node, time.time())
            duration_ms = int(duration * 1000)
        else:
            duration_ms = None
        
        if token_usage:
            self.token_usage[node] = token_usage
        
        await self.ws_send({
            "type": "debug_log",
            "request_id": self.request_id,
            "category": "node",
            "timestamp": int(time.time() * 1000),
            "data": {
                "node": node,
                "stage": stage,
                "message": message,
                "token_usage": token_usage,
                "duration_ms": duration_ms,
            }
        })
    
    async def log_thinking(self, node: str, reasoning_delta: str):
        """记录思考内容"""
        await self.ws_send({
            "type": "debug_log",
            "request_id": self.request_id,
            "category": "thinking",
            "timestamp": int(time.time() * 1000),
            "data": {
                "node": node,
                "reasoning_delta": reasoning_delta,
            }
        })
    
    async def log_tool(
        self,
        node: str,
        tool_name: str,
        input_data: Any,
        output: str,
        duration_ms: float,
    ):
        """记录工具调用"""
        await self.ws_send({
            "type": "debug_log",
            "request_id": self.request_id,
            "category": "tool",
            "timestamp": int(time.time() * 1000),
            "data": {
                "node": node,
                "tool_name": tool_name,
                "input": str(input_data)[:500],  # 截断大输入
                "output": output[:500],
                "duration_ms": duration_ms,
            }
        })
    
    async def log_error(self, node: str, error: str, stack: str | None = None):
        """记录错误"""
        await self.ws_send({
            "type": "debug_log",
            "request_id": self.request_id,
            "category": "error",
            "timestamp": int(time.time() * 1000),
            "data": {
                "node": node,
                "error": error,
                "stack": stack,
            }
        })
    
    async def send_final_metrics(self, retry_count: int, validation_results: list):
        """发送最终统计"""
        total_duration = time.time() - self.start_time
        
        total_tokens = {
            "input": sum(t.get("input", 0) for t in self.token_usage.values()),
            "output": sum(t.get("output", 0) for t in self.token_usage.values()),
        }
        total_tokens["total"] = total_tokens["input"] + total_tokens["output"]
        
        validation_errors = sum(
            1 for r in validation_results if r.has_error
        )
        
        await self.ws_send({
            "type": "session_metrics",
            "request_id": self.request_id,
            "metrics": {
                "total_duration_ms": int(total_duration * 1000),
                "node_durations": {
                    node: int((time.time() - start) * 1000)
                    for node, start in self.node_timings.items()
                },
                "total_tokens": total_tokens,
                "node_tokens": self.token_usage,
                "rag_queries": self.rag_stats["queries"],
                "rag_cache_hits": self.rag_stats["cache_hits"],
                "rag_total_chunks": self.rag_stats["total_chunks"],
                "retry_count": retry_count,
                "validation_steps": len(validation_results),
                "validation_errors": validation_errors,
            }
        })
```

```python
# wild-server/app/agent/nodes/skeleton_node.py
async def skeleton_generator(
    state: GenerationState,
    debug_logger: DebugLogger | None = None,
) -> dict:
    """生成骨架，带完整日志"""
    
    if debug_logger:
        await debug_logger.log_node_stage(
            "skeleton_generator",
            "start",
            "开始生成建筑骨架"
        )
    
    # ── RAG 检索 ──
    rag_start = time.time()
    queries = [
        SpecQuery(state["user_message"], {"doc_type": "building_type"}),
        SpecQuery(state["user_message"], {"entity_type": "wall"}),
    ]
    spec_text = spec_loader.load_many(queries, per_query=2)
    rag_duration = (time.time() - rag_start) * 1000
    
    if debug_logger:
        # 记录每个查询的命中结果
        for i, query in enumerate(queries):
            hits = spec_loader.last_results[i*2:(i+1)*2]  # 每个查询取 2 个
            await debug_logger.log_rag(
                node="skeleton_generator",
                query=query.text,
                filter=query.metadata_filter or {},
                hits=hits,
                duration_ms=rag_duration / len(queries),
                cache_hit=False,  # 可从 Chroma 缓存状态获取
            )
    
    # ── LLM 调用 ──
    if debug_logger:
        await debug_logger.log_node_stage(
            "skeleton_generator",
            "llm_start",
            "调用 LLM 生成骨架"
        )
    
    # 思考模式的 callback
    async def on_thinking_delta(delta: str):
        if debug_logger:
            await debug_logger.log_thinking("skeleton_generator", delta)
    
    llm_start = time.time()
    result = await agent.ainvoke(
        {"messages": [{"role": "user", "content": state["user_message"]}]},
        config={"callbacks": [ThinkingCallback(on_thinking_delta)]} if thinking_mode else None,
    )
    llm_duration = (time.time() - llm_start) * 1000
    
    # 提取 token usage（从 LangChain response）
    token_usage = None
    if hasattr(result, "response_metadata"):
        usage = result.response_metadata.get("usage", {})
        token_usage = {
            "input": usage.get("prompt_tokens", 0),
            "output": usage.get("completion_tokens", 0),
            "total": usage.get("total_tokens", 0),
        }
    
    if debug_logger:
        await debug_logger.log_node_stage(
            "skeleton_generator",
            "llm_done",
            f"LLM 完成，耗时 {llm_duration:.0f}ms",
            token_usage=token_usage,
        )
    
    # ── 提取和校验 ──
    blueprint = extract_blueprint_from_text(result["messages"][-1].content)
    
    # ── 工具调用 ──
    tool_start = time.time()
    bbox_result = get_wall_bounding_box(blueprint)
    tool_duration = (time.time() - tool_start) * 1000
    
    if debug_logger:
        await debug_logger.log_tool(
            node="skeleton_generator",
            tool_name="get_wall_bounding_box",
            input_data={"wall_count": len([e for e in blueprint["geometry"]["elements"] if e["type"] == "wall"])},
            output=str(bbox_result)[:200],
            duration_ms=tool_duration,
        )
    
    if debug_logger:
        await debug_logger.log_node_stage(
            "skeleton_generator",
            "complete",
            "骨架生成完成"
        )
    
    return {
        "skeleton_blueprint": blueprint,
        "skeleton_summary": _build_summary(blueprint),
        "wall_bounding_box": bbox_result,
    }
```

```typescript
  node_durations: Record<string, number>

    input: number
    output: number
    total: number

  validation_errors: number

const debugLogs = ref<DebugLog[]>([])
const sessionMetrics = ref<SessionMetrics | null>(null)
const debugPanelVisible = ref(false)

function addDebugLog(log: DebugLog) {
  debugLogs.value.push(log)
}

function clearDebugLogs() {
  debugLogs.value = []
  sessionMetrics.value = null
}

function setSessionMetrics(metrics: SessionMetrics) {
  sessionMetrics.value = metrics
}

function toggleDebugPanel() {
  debugPanelVisible.value = !debugPanelVisible.value
}

export function useAgentStore() {
  return {
    // ...现有的
    debugLogs,
    sessionMetrics,
    debugPanelVisible,
    addDebugLog,
    clearDebugLogs,
    setSessionMetrics,
    toggleDebugPanel,
  }
}
```

```vue
import { computed } from 'vue'
import { useAgentStore } from '@/stores/agentStore'

const categoryIcons = {
  rag: '🔎',
  node: '🔷',
  thinking: '💭',
  tool: '🔧',
  error: '❌',
}

const categoryLabels = {
  rag: 'RAG 检索',

  thinking: '思考',
  tool: '工具',
  error: '错误',
}

const nodeLabels = {

  door_generator: '门生成',
  window_generator: '窗生成',
  roof_generator: '屋顶生成',
  // ...
}

const maxDuration = computed(() => {
  if (!agentStore.sessionMetrics) return 1
  return Math.max(...Object.values(agentStore.sessionMetrics.node_durations))
})

function formatDuration(ms: number): string {
  if (ms < 1000) return `${ms}ms`
  return `${(ms / 1000).toFixed(1)}s`
}

function formatTime(timestamp: number): string {
  return new Date(timestamp).toLocaleTimeString('zh-CN', {
    hour: '2-digit',
    minute: '2-digit',
    second: '2-digit',
    fractionalSecondDigits: 3,
  })
}

function exportLogs() {
  const data = {
    metrics: agentStore.sessionMetrics,
    logs: agentStore.debugLogs,
    timestamp: Date.now(),
  }
  const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' })
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = `debug-log-${Date.now()}.json`
  a.click()
  URL.revokeObjectURL(url)
}
```
